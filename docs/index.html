Follow one program through the compiler

This is the recursive factorial example from the repository and a small part of the LLVM IR generated from it. 

program factorial;

var n : integer;

function fact (x : integer) : integer;
begin
    if x <= 1 then
        result := 1
    else
        result := x * fact(x - 1)
end;

begin
    n := readInteger();
    writeString("The factorial of ");
    writeInteger(n);
    writeString(" is ");
    writeInteger(fact(n));
    writeString("\n")
end.
define i64 @pcl_fact_0(ptr %static_link, i64 %arg0) {
entry:
  %tmp3 = alloca i64
  store i64 %arg0, ptr %tmp3
  %tmp8 = load i64, ptr %tmp3
  %tmp9 = icmp sle i64 %tmp8, 1
  br i1 %tmp9, label %if_then_1, label %if_else_2

if_else_2:
  %tmp12 = sub i64 %tmp11, 1
  %tmp15 = call i64 @pcl_fact_0(ptr %tmp14, i64 %tmp12)
  %tmp16 = mul i64 %tmp10, %tmp15
  store i64 %tmp16, ptr %tmp2
  br label %return_0

return_0:
  %tmp17 = load i64, ptr %tmp2
  ret i64 %tmp17
}

Under the hood

The parts that required the most work

01

Semantic analysis

Type checking, lexical scopes, forward declarations, lvalue validation and precise function-call checks.

02

Nested functions

Pascal-style lexical scoping, variable capture and static links between nested stack frames.

03

LLVM code generation

Typed AST lowering, short-circuit control flow, memory operations and native linking through Clang.
