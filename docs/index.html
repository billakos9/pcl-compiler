factorial.pcl A real example 
function fact (x : integer) : integer;
begin
    if x <= 1 then
        result := 1
    else
        result := x * fact(x - 1)
end;
; Recursive call — generated IR excerpt
%tmp12 = sub i64 %tmp11, 1
%tmp15 = call i64 @pcl_fact_0(
  ptr %tmp14, i64 %tmp12)
%tmp16 = mul i64 %tmp10, %tmp15
store i64 %tmp16, ptr %tmp2
br label %return_0
